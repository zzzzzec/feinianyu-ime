set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# Keep inference optimized in an otherwise debuggable APK.
add_compile_options(-O3)
add_compile_definitions(NDEBUG)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
set(GGML_CPU_REPACK ON CACHE BOOL "" FORCE)
# Never raise the baseline ISA of the APK. ggml checks CPU features before loading a variant.
if(ANDROID_ABI STREQUAL "arm64-v8a")
    set(KLEIDIAI_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../third_party/kleidiai")
    if(NOT EXISTS "${KLEIDIAI_SOURCE}/.source-sha256")
        message(FATAL_ERROR "Run scripts/fetch-kleidiai.ps1 before building arm64-v8a")
    endif()
    file(READ "${KLEIDIAI_SOURCE}/.source-sha256" KLEIDIAI_SOURCE_SHA)
    string(STRIP "${KLEIDIAI_SOURCE_SHA}" KLEIDIAI_SOURCE_SHA)
    if(NOT KLEIDIAI_SOURCE_SHA STREQUAL "a62d13f9d091badfc286bf08cbba116f834cd330ae28ba1c0e48d0c53a7dc333")
        message(FATAL_ERROR "Unexpected KleidiAI source version")
    endif()
    set(FETCHCONTENT_SOURCE_DIR_KLEIDIAI "${KLEIDIAI_SOURCE}" CACHE PATH "" FORCE)
    set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
    set(GGML_CPU_KLEIDIAI ON CACHE BOOL "" FORCE)
else()
    set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
    set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
endif()
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)

    ../../../../../third_party/suzu/model/suzu_model.cpp)
